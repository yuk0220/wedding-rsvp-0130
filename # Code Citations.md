```html

<html lang="ko">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>청첩장</title>
    <link rel="stylesheet" href="styles.css">
</head>
<body>
    <div class="container">
        <div class="invitation-card">
            <!-- 헤더 -->
            <div class="header">
                <div class="decorative-line"></div>
                <h1>모시는 글</h1>

            </div>

            <div class="greeting">

            </div>

            <!-- 신랑신부 정보 -->
